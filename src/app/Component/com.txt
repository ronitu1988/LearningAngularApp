CREATE OR REPLACE PACKAGE BODY PHT_MATERIAL_APPROVAL_PKG AS

    PROCEDURE GET_APPROVAL_QUEUE (
        P_CURSOR OUT SYS_REFCURSOR
    ) IS
    BEGIN
        OPEN P_CURSOR FOR
            SELECT
                GROUP_ID,
                STATUS,
                APL,
                CREATED_DT,
                CREATED_BY
            FROM (
                SELECT
                    GROUP_ID,
                    MAX(STATUS) AS STATUS,
                    MAX(APL) AS APL,
                    MAX(CREATED_DT) AS CREATED_DT,
                    MAX(CREATED_BY) AS CREATED_BY
                FROM PHT_MATERIAL_REQUEST_SUBMISSION
                WHERE NVL(IS_DELETED, 0) = 0
                  AND STATUS = 'Submitted'
                GROUP BY GROUP_ID
            )
            ORDER BY CREATED_DT DESC;
    END GET_APPROVAL_QUEUE;


    PROCEDURE GET_REQUEST_GROUP_DETAILS (
        P_GROUP_ID IN NUMBER,
        P_CURSOR   OUT SYS_REFCURSOR
    ) IS
    BEGIN
        OPEN P_CURSOR FOR
            SELECT
                S.GROUP_ID,
                S.APL,
                S.HULL_ID,
                S.NIIN_ID,
                S.NIIN_NOMENCLATURE,
                S.PART_NUMBER,
                S.QTY_REQUESTED,
                S.STATUS,

                A.HULL_ID AS APPROVAL_HULL_ID,
                A.ON_LOCATION,
                A.APPROVED_QTY,
                A.APPROVAL_ACTION,
                A.APPROVAL_NOTE
            FROM PHT_MATERIAL_REQUEST_SUBMISSION S
            LEFT JOIN PHT_MATERIAL_REQUEST_APPROVAL A
              ON A.GROUP_ID = S.GROUP_ID
             AND A.NIIN_ID  = S.NIIN_ID
             AND NVL(A.IS_DELETED, 0) = 0
            WHERE S.GROUP_ID = P_GROUP_ID
              AND NVL(S.IS_DELETED, 0) = 0
            ORDER BY S.NIIN_ID;
    END GET_REQUEST_GROUP_DETAILS;


    PROCEDURE GET_HULL_LIST (
        P_CURSOR OUT SYS_REFCURSOR
    ) IS
    BEGIN
        OPEN P_CURSOR FOR
            SELECT DISTINCT
                   HULL_ID,
                   HULL_NAME
            FROM PHT_HULL_ELIGIBILITY
            ORDER BY HULL_NAME;
    END GET_HULL_LIST;


    PROCEDURE GET_ON_LOCATION_LIST (
        P_CURSOR OUT SYS_REFCURSOR
    ) IS
    BEGIN
        OPEN P_CURSOR FOR
            SELECT LOCATION_CODE,
                   LOCATION_NAME
            FROM PHT_ON_LOCATION_LOOKUP
            ORDER BY LOCATION_NAME;
    END GET_ON_LOCATION_LIST;


    PROCEDURE SAVE_APPROVAL (
        P_GROUP_ID          IN NUMBER,
        P_HULL_ID           IN VARCHAR2,
        P_ON_LOCATION       IN VARCHAR2,
        P_ACTION            IN VARCHAR2,
        P_NOTE              IN VARCHAR2,
        P_ACTION_BY         IN VARCHAR2,

        P_NIIN_IDS          IN T_VARCHAR2_TAB,
        P_APPROVED_QTYS     IN T_NUMBER_TAB,

        P_MESSAGE           OUT VARCHAR2
    ) IS
    BEGIN
        IF P_NIIN_IDS.COUNT = 0 THEN
            P_MESSAGE := 'No NIIN rows provided.';
            RETURN;
        END IF;

        FOR I IN 1 .. P_NIIN_IDS.COUNT LOOP
            MERGE INTO PHT_MATERIAL_REQUEST_APPROVAL T
            USING (
                SELECT
                    P_GROUP_ID              AS GROUP_ID,
                    P_NIIN_IDS(I)           AS NIIN_ID
                FROM DUAL
            ) S
            ON (
                T.GROUP_ID = S.GROUP_ID
                AND T.NIIN_ID = S.NIIN_ID
            )
            WHEN MATCHED THEN
                UPDATE SET
                    T.HULL_ID         = P_HULL_ID,
                    T.ON_LOCATION     = P_ON_LOCATION,
                    T.APPROVED_QTY    = P_APPROVED_QTYS(I),
                    T.APPROVAL_ACTION = P_ACTION,
                    T.APPROVAL_NOTE   = P_NOTE,
                    T.ACTION_BY       = P_ACTION_BY,
                    T.ACTION_DATE     = SYSTIMESTAMP,
                    T.UPDATE_DT       = SYSTIMESTAMP,
                    T.UPDATE_BY       = P_ACTION_BY,
                    T.IS_DELETED      = 0
            WHEN NOT MATCHED THEN
                INSERT (
                    GROUP_ID,
                    NIIN_ID,
                    HULL_ID,
                    ON_LOCATION,
                    APPROVED_QTY,
                    APPROVAL_ACTION,
                    APPROVAL_NOTE,
                    ACTION_BY,
                    ACTION_DATE,
                    IS_DELETED,
                    CREATED_DT,
                    CREATED_BY
                )
                VALUES (
                    P_GROUP_ID,
                    P_NIIN_IDS(I),
                    P_HULL_ID,
                    P_ON_LOCATION,
                    P_APPROVED_QTYS(I),
                    P_ACTION,
                    P_NOTE,
                    P_ACTION_BY,
                    SYSTIMESTAMP,
                    0,
                    SYSTIMESTAMP,
                    P_ACTION_BY
                );
        END LOOP;

        UPDATE PHT_MATERIAL_REQUEST_SUBMISSION
           SET STATUS    = CASE
                             WHEN UPPER(P_ACTION) = 'APPROVED' THEN 'Approved'
                             WHEN UPPER(P_ACTION) = 'DENIED'   THEN 'Denied'
                             ELSE STATUS
                           END,
               UPDATE_DT = SYSTIMESTAMP,
               UPDATE_BY = P_ACTION_BY
         WHERE GROUP_ID = P_GROUP_ID
           AND NVL(IS_DELETED, 0) = 0;

        COMMIT;
        P_MESSAGE := 'Approval saved successfully.';
    EXCEPTION
        WHEN OTHERS THEN
            ROLLBACK;
            P_MESSAGE := 'Approval save failed: ' || SQLERRM;
            RAISE;
    END SAVE_APPROVAL;

END PHT_MATERIAL_APPROVAL_PKG;
/

    {
        var results = new List<ApprovalQueueDto>();

        await using var conn = new OracleConnection(_connectionString);
        await conn.OpenAsync();

        await using var cmd = new OracleCommand("PHT_MATERIAL_APPROVAL_PKG.GET_APPROVAL_QUEUE", conn)
        {
            CommandType = CommandType.StoredProcedure
        };

        cmd.Parameters.Add("P_CURSOR", OracleDbType.RefCursor, ParameterDirection.Output);

        await using var reader = await cmd.ExecuteReaderAsync();
        while (await reader.ReadAsync())
        {
            results.Add(new ApprovalQueueDto
            {
                GroupId = Convert.ToDecimal(reader["GROUP_ID"]),
                Status = reader["STATUS"]?.ToString(),

                CreatedDt = reader["CREATED_DT"] == DBNull.Value ? null : Convert.ToDateTime(reader["CREATED_DT"]),

    public async Task<string> SaveApprovalAsync(SaveApprovalRequestDto dto, string actionBy)
    {
        var validLines = dto.Lines.ToList();

        await using var conn = new OracleConnection(_connectionString);
        await conn.OpenAsync();

        await using var cmd = new OracleCommand("PHT_MATERIAL_APPROVAL_PKG.SAVE_APPROVAL", conn)
        {
            CommandType = CommandType.StoredProcedure
        };

        cmd.Parameters.Add("P_GROUP_ID", OracleDbType.Decimal).Value = dto.GroupId;
        cmd.Parameters.Add("P_HULL_ID", OracleDbType.Varchar2).Value = dto.HullId;
        cmd.Parameters.Add("P_ON_LOCATION", OracleDbType.Varchar2).Value = dto.OnLocation;
        cmd.Parameters.Add("P_ACTION", OracleDbType.Varchar2).Value = dto.Action;
        cmd.Parameters.Add("P_NOTE", OracleDbType.Varchar2).Value = dto.Note;
        cmd.Parameters.Add("P_ACTION_BY", OracleDbType.Varchar2).Value = actionBy;

        cmd.Parameters.Add(BuildStringArray("P_NIIN_IDS", validLines.Select(x => x.NiinId).ToArray()));
        cmd.Parameters.Add(BuildNumberArray("P_APPROVED_QTYS", validLines.Select(x => x.ApprovedQty).ToArray()));

        var messageParam = new OracleParameter("P_MESSAGE", OracleDbType.Varchar2, 4000)
        {
            Direction = ParameterDirection.Output
        };
        cmd.Parameters.Add(messageParam);

        await cmd.ExecuteNonQueryAsync();
        return messageParam.Value?.ToString() ?? "Completed.";
    }

    private static OracleParameter BuildStringArray(string name, string[] values)
    {
        return new OracleParameter(name, OracleDbType.Varchar2)
        {
            CollectionType = OracleCollectionType.PLSQLAssociativeArray,
            Direction = ParameterDirection.Input,
            Size = values.Length,
            Value = values
        };
    }

    private static OracleParameter BuildNumberArray(string name, decimal[] values)
    {
        return new OracleParameter(name, OracleDbType.Decimal)
        {
            CollectionType = OracleCollectionType.PLSQLAssociativeArray,
            Direction = ParameterDirection.Input,
            Size = values.Length,
            Value = values
        };
    }
}

import { CommonModule } from '@angular/common';
import { Component, OnInit, inject } from '@angular/core';
import { FormArray, FormBuilder, FormGroup, ReactiveFormsModule, Validators } from '@angular/forms';
import { ApprovalService } from './approval.service';
import {
  ApprovalGroupDetail,
  ApprovalQueueItem,
  HullLookup,
  OnLocationLookup,
  SaveApprovalRequest
} from './approval.models';

@Component({
  selector: 'app-approval',
  standalone: true,
  imports: [CommonModule, ReactiveFormsModule],
  templateUrl: './approval.component.html',
  styleUrl: './approval.component.scss'
})
export class ApprovalComponent implements OnInit {
  private fb = inject(FormBuilder);
  private service = inject(ApprovalService);

  queue: ApprovalQueueItem[] = [];
  hulls: HullLookup[] = [];
  locations: OnLocationLookup[] = [];
  selectedGroupId: number | null = null;

  form = this.fb.group({
    groupId: [0, Validators.required],
    hullId: ['', Validators.required],
    onLocation: ['', Validators.required],
    note: [''],
    lines: this.fb.array([])
  });

  get lines(): FormArray<FormGroup> {
    return this.form.get('lines') as FormArray<FormGroup>;
  }

  ngOnInit(): void {
    this.loadQueue();
    this.loadHulls();
    this.loadLocations();
  }

  loadQueue(): void {
    this.service.getQueue().subscribe({
      next: data => this.queue = data ?? [],
      error: err => console.error('Queue load failed', err)
    });
  }

  loadHulls(): void {
    this.service.getHulls().subscribe({
      next: data => this.hulls = data ?? [],
      error: err => console.error('Hull load failed', err)
    });
  }

  loadLocations(): void {
    this.service.getLocations().subscribe({
      next: data => this.locations = data ?? [],
      error: err => console.error('Location load failed', err)
    });
  }

  selectGroup(item: ApprovalQueueItem): void {
    this.selectedGroupId = item.groupId;
    this.form.patchValue({ groupId: item.groupId });
    this.lines.clear();

    this.service.getGroup(item.groupId).subscribe({
      next: rows => {
        (rows ?? []).forEach(row => {
          this.lines.push(
            this.fb.group({
              niinId: [row.niinId ?? ''],
              niinNomenclature: [row.niinNomenclature ?? ''],
              partNumber: [row.partNumber ?? ''],
              qtyRequested: [row.qtyRequested ?? 0],
              approvedQty: [row.approvedQty ?? row.qtyRequested ?? 0, [Validators.required, Validators.min(0)]]
            })
          );
        });
      },
      error: err => console.error('Group load failed', err)
    });
  }

  approve(): void {
    this.save('Approved');
  }

  deny(): void {
    this.save('Denied');
  }

  private save(action: string): void {
    if (this.form.invalid) {
      this.form.markAllAsTouched();
      return;
    }

    const raw = this.form.getRawValue();

    const payload: SaveApprovalRequest = {
      groupId: Number(raw.groupId),
      hullId: raw.hullId ?? '',
      onLocation: raw.onLocation ?? '',
      action,
      note: raw.note ?? '',
      lines: (raw.lines ?? []).map((x: any) => ({
        niinId: x.niinId,
        approvedQty: Number(x.approvedQty ?? 0)
      }))
    };

    this.service.save(payload).subscribe({
      next: res => {
        alert(res?.message ?? 'Saved.');
        this.lines.clear();
        this.selectedGroupId = null;
        this.form.reset({ groupId: 0, hullId: '', onLocation: '', note: '' });
        this.loadQueue();
      },
      error: err => {
        console.error('Save failed', err);
        alert('Save failed.');
      }
    });
  }
}
