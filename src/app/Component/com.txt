CREATE OR REPLACE PACKAGE BODY PRTHRVST.PKG_PHT_REMOVAL_STATUS AS

    PROCEDURE CREATE_UPLOAD_BATCH (
        P_FILE_NAME   IN  VARCHAR2,
        P_FILE_TYPE   IN  VARCHAR2,
        P_UPLOADED_BY IN  VARCHAR2,
        O_BATCH_ID    OUT NUMBER
    ) AS
    BEGIN
        INSERT INTO PRTHRVST.PHT_REMOVAL_UPLOAD_BATCH (
            FILE_NAME,
            FILE_TYPE,
            UPLOADED_BY
        )
        VALUES (
            P_FILE_NAME,
            P_FILE_TYPE,
            P_UPLOADED_BY
        )
        RETURNING BATCH_ID INTO O_BATCH_ID;
    END;

    PROCEDURE UPSERT_REMOVAL_ROW (
        P_BATCH_ID        IN NUMBER,
        P_HULL_NO         IN VARCHAR2,
        P_ERLID           IN NUMBER,
        P_APL             IN VARCHAR2,
        P_APL_NOMEN       IN VARCHAR2,
        P_SAC_NOMEN       IN VARCHAR2,
        P_ONBOARD_LOCATION IN VARCHAR2,
        P_PART_NUMBER     IN VARCHAR2,
        P_NSN             IN VARCHAR2,
        P_DWG_NO          IN VARCHAR2,
        P_QTY             IN NUMBER,
        P_CAS_REP_IND     IN VARCHAR2,
        P_IDENT           IN VARCHAR2,
        P_REMOVED         IN VARCHAR2,
        P_DATE_REMOVED    IN DATE,
        P_SYSTEM_ID       IN VARCHAR2,
        P_FUND_CODE       IN VARCHAR2,
        P_DOCUMENT_NO     IN VARCHAR2,
        P_REMOVED_BY      IN VARCHAR2,
        P_REMARKS         IN VARCHAR2,
        P_UPDATED_BY      IN VARCHAR2,
        O_ACTION          OUT VARCHAR2
    ) AS
        V_ID NUMBER;
    BEGIN
        BEGIN
            SELECT ID
            INTO V_ID
            FROM PRTHRVST.PHT_MATERIAL_REQUEST_SUBMISSION
            WHERE TO_CHAR(HULL_ID) = P_HULL_NO
              AND NVL(APL, 'X') = NVL(P_APL, 'X')
              AND NVL(PART_NUMBER, 'X') = NVL(P_PART_NUMBER, 'X')
              AND NVL(NSN, 'X') = NVL(P_NSN, 'X')
              AND NVL(APPROVED_QTY, QTY_REQUESTED) = P_QTY
              AND IS_DELETED = 0
              FETCH FIRST 1 ROW ONLY;

            UPDATE PRTHRVST.PHT_MATERIAL_REQUEST_SUBMISSION
            SET
                ERLID = P_ERLID,
                APL = P_APL,
                APL_NOMEN = P_APL_NOMEN,
                SAC_NOMEN = P_SAC_NOMEN,
                ON_LOCATION = NVL(P_ONBOARD_LOCATION, ON_LOCATION),
                PART_NUMBER = P_PART_NUMBER,
                NSN = P_NSN,
                DWG_NO = P_DWG_NO,
                CAS_REP_IND = P_CAS_REP_IND,
                IDENT = P_IDENT,
                SHIPYARD_REMOVED = P_REMOVED,
                DATE_REMOVED = P_DATE_REMOVED,
                SYSTEM_ID = P_SYSTEM_ID,
                FUND_CODE = P_FUND_CODE,
                DOCUMENT_NO = P_DOCUMENT_NO,
                REMOVED_BY = P_REMOVED_BY,
                SHIPYARD_REMARKS = P_REMARKS,
                SHIPYARD_STATUS =
                    CASE
                        WHEN P_REMOVED = 'Y' THEN 'Removed'
                        WHEN P_DATE_REMOVED IS NOT NULL THEN 'Removed'
                        ELSE 'In Process'
                    END,
                STATUS =
                    CASE
                        WHEN P_REMOVED = 'Y' OR P_DATE_REMOVED IS NOT NULL THEN 'Removed'
                        ELSE STATUS
                    END,
                REMOVAL_UPLOAD_BATCH_ID = P_BATCH_ID,
                REMOVAL_SOURCE = 'UPLOAD',
                REMOVAL_MATCH_STATUS = 'MATCHED',
                REMOVAL_UPDATED_DT = SYSTIMESTAMP,
                REMOVAL_UPDATED_BY = P_UPDATED_BY,
                UPDATED_DT = SYSTIMESTAMP,
                UPDATED_BY = P_UPDATED_BY
            WHERE ID = V_ID;

            O_ACTION := 'MATCHED';

        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                INSERT INTO PRTHRVST.PHT_MATERIAL_REQUEST_SUBMISSION (
                    GROUP_ID,
                    STATUS,
                    HULL_ID,
                    RIC_CODE,
                    NIIN_ID,
                    QTY_REQUESTED,
                    APPROVED_QTY,
                    ERLID,
                    APL,
                    APL_NOMEN,
                    SAC_NOMEN,
                    ON_LOCATION,
                    PART_NUMBER,
                    NSN,
                    DWG_NO,
                    CAS_REP_IND,
                    IDENT,
                    SHIPYARD_REMOVED,
                    DATE_REMOVED,
                    SYSTEM_ID,
                    FUND_CODE,
                    DOCUMENT_NO,
                    REMOVED_BY,
                    SHIPYARD_REMARKS,
                    SHIPYARD_STATUS,
                    REMOVAL_UPLOAD_BATCH_ID,
                    REMOVAL_SOURCE,
                    REMOVAL_MATCH_STATUS,
                    CREATED_BY,
                    CREATED_DT,
                    UPDATED_BY,
                    UPDATED_DT,
                    IS_DELETED
                )
                VALUES (
                    P_BATCH_ID,
                    'Shipyard Only',
                    TO_NUMBER(NULLIF(P_HULL_NO, '')),
                    'UPLOAD',
                    0,
                    P_QTY,
                    P_QTY,
                    P_ERLID,
                    P_APL,
                    P_APL_NOMEN,
                    P_SAC_NOMEN,
                    P_ONBOARD_LOCATION,
                    P_PART_NUMBER,
                    P_NSN,
                    P_DWG_NO,
                    P_CAS_REP_IND,
                    P_IDENT,
                    P_REMOVED,
                    P_DATE_REMOVED,
                    P_SYSTEM_ID,
                    P_FUND_CODE,
                    P_DOCUMENT_NO,
                    P_REMOVED_BY,
                    P_REMARKS,
                    CASE
                        WHEN P_REMOVED = 'Y' THEN 'Removed'
                        WHEN P_DATE_REMOVED IS NOT NULL THEN 'Removed'
                        ELSE 'Shipyard Only'
                    END,
                    P_BATCH_ID,
                    'UPLOAD',
                    'INSERTED',
                    P_UPDATED_BY,
                    SYSTIMESTAMP,
                    P_UPDATED_BY,
                    SYSTIMESTAMP,
                    0
                );

                O_ACTION := 'INSERTED';
        END;
    END;

    PROCEDURE GET_SUMMARY (
        P_BATCH_ID IN NUMBER,
        O_CURSOR   OUT SYS_REFCURSOR
    ) AS
    BEGIN
        OPEN O_CURSOR FOR
            SELECT
                COUNT(*) AS TOTAL_ERL_REQUESTS,
                SUM(CASE WHEN SHIPYARD_STATUS = 'In Process' THEN 1 ELSE 0 END) AS IN_PROCESS,
                SUM(CASE WHEN SHIPYARD_STATUS = 'Removed' THEN 1 ELSE 0 END) AS REMOVED,
                SUM(CASE WHEN REMOVAL_MATCH_STATUS = 'INSERTED' THEN 1 ELSE 0 END) AS DISCREPANCIES
            FROM PRTHRVST.PHT_MATERIAL_REQUEST_SUBMISSION
            WHERE IS_DELETED = 0
              AND (P_BATCH_ID IS NULL OR REMOVAL_UPLOAD_BATCH_ID = P_BATCH_ID);
    END;

    PROCEDURE GET_GRID (
        P_BATCH_ID IN NUMBER,
        P_SYSTEM_ID IN VARCHAR2,
        P_STATUS IN VARCHAR2,
        P_SEARCH IN VARCHAR2,
        O_CURSOR OUT SYS_REFCURSOR
    ) AS
    BEGIN
        OPEN O_CURSOR FOR
            SELECT
                ID,
                HULL_ID,
                ERLID,
                APL,
                APL_NOMEN,
                SAC_NOMEN,
                ON_LOCATION,
                PART_NUMBER,
                NSN,
                DWG_NO,
                QTY_REQUESTED,
                APPROVED_QTY,
                SHIPYARD_STATUS,
                SHIPYARD_REMARKS,
                DOCUMENT_NO,
                REMOVED_BY,
                DATE_REMOVED,
                SYSTEM_ID,
                FUND_CODE,
                REMOVAL_MATCH_STATUS,
                REMOVAL_UPLOAD_BATCH_ID
            FROM PRTHRVST.PHT_MATERIAL_REQUEST_SUBMISSION
            WHERE IS_DELETED = 0
              AND (P_BATCH_ID IS NULL OR REMOVAL_UPLOAD_BATCH_ID = P_BATCH_ID)
              AND (P_SYSTEM_ID IS NULL OR SYSTEM_ID = P_SYSTEM_ID)
              AND (P_STATUS IS NULL OR SHIPYARD_STATUS = P_STATUS)
              AND (
                    P_SEARCH IS NULL
                    OR TO_CHAR(ERLID) LIKE '%' || P_SEARCH || '%'
                    OR UPPER(APL_NOMEN) LIKE '%' || UPPER(P_SEARCH) || '%'
                    OR UPPER(NSN) LIKE '%' || UPPER(P_SEARCH) || '%'
                    OR UPPER(PART_NUMBER) LIKE '%' || UPPER(P_SEARCH) || '%'
                  )
            ORDER BY ERLID;
    END;

END PKG_PHT_REMOVAL_STATUS;

    public async Task<RemovalSummaryDto> GetSummaryAsync(decimal? batchId)
    {
        await using var conn = GetConnection();
        await conn.OpenAsync();

        await using var cmd = conn.CreateCommand();
        cmd.CommandText = "PRTHRVST.PKG_PHT_REMOVAL_STATUS.GET_SUMMARY";
        cmd.CommandType = CommandType.StoredProcedure;

        cmd.Parameters.Add("P_BATCH_ID", OracleDbType.Decimal).Value = (object?)batchId ?? DBNull.Value;
        cmd.Parameters.Add("O_CURSOR", OracleDbType.RefCursor).Direction = ParameterDirection.Output;

        await using var reader = await cmd.ExecuteReaderAsync();

        if (await reader.ReadAsync())
        {
            return new RemovalSummaryDto
            {
                TotalErlRequests = reader.GetDecimalSafe("TOTAL_ERL_REQUESTS"),
                InProcess = reader.GetDecimalSafe("IN_PROCESS"),
                Removed = reader.GetDecimalSafe("REMOVED"),
                Discrepancies = reader.GetDecimalSafe("DISCREPANCIES")
            };
        }

        return new RemovalSummaryDto();
    }

    public async Task<RemovalUploadResultDto> UploadExcelAsync(IFormFile file, string uploadedBy)
    {
        var rows = new List<ParsedRemovalRowDto>();

        using var stream = file.OpenReadStream();
        using var workbook = new XLWorkbook(stream);
        var ws = workbook.Worksheets.First();
        var headerRow = ws.FirstRowUsed();

        var headers = headerRow.Cells().ToDictionary(
            c => c.GetString().Trim(),
            c => c.Address.ColumnNumber,
            StringComparer.OrdinalIgnoreCase
        );

        foreach (var row in ws.RowsUsed().Skip(1))
        {
            rows.Add(new ParsedRemovalRowDto
            {
                HullNo = GetCell(row, headers, "HullNo"),
                Erlid = ToDecimal(GetCell(row, headers, "ERLID")),
                Apl = GetCell(row, headers, "APL"),
                AplNomen = GetCell(row, headers, "APLNomen"),
                SacNomen = GetCell(row, headers, "SacNomen"),
                OnBoardLocation = GetCell(row, headers, "OnBoardLocation"),
                PartNumber = GetCell(row, headers, "PartNumber"),
                Nsn = GetCell(row, headers, "NSN"),
                DwgNo = GetCell(row, headers, "DwgNo"),
                Qty = ToDecimal(GetCell(row, headers, "Qty")),
                CasRepInd = GetCell(row, headers, "CasRepInd"),
                Ident = GetCell(row, headers, "Ident"),
                Removed = GetCell(row, headers, "Removed"),
                DateRemoved = ToDate(GetCell(row, headers, "DateRemoved")),
                SystemId = GetCell(row, headers, "SystemID"),
                FundCode = GetCell(row, headers, "FundCode"),
                DocumentNo = GetCell(row, headers, "Document"),
                Remarks = GetCell(row, headers, "Remarks")
            });
        }

        return await SaveRowsAsync(file.FileName, "EXCEL", uploadedBy, rows);
    }

    public async Task<RemovalUploadResultDto> UploadTxtAsync(IFormFile file, string uploadedBy)
    {
        var rows = new List<ParsedRemovalRowDto>();

        using var reader = new StreamReader(file.OpenReadStream());
        var text = await reader.ReadToEndAsync();

        var lines = text.Split(Environment.NewLine, StringSplitOptions.RemoveEmptyEntries);

        foreach (var line in lines)
        {
            var cleaned = Regex.Replace(line.Trim(), @"\s+", " ");
            var tokens = cleaned.Split(' ', StringSplitOptions.RemoveEmptyEntries);

            if (tokens.Length < 10)
                continue;

            var firstValue = tokens[0].Replace(".00", "");
            var hullNo = firstValue.Length >= 3 ? firstValue[..3] : null;
            var erlidText = firstValue.Length > 3 ? firstValue[3..] : null;

            var yQtyToken = tokens.FirstOrDefault(x => Regex.IsMatch(x, @"^Y\d+\.00$", RegexOptions.IgnoreCase));
            var qty = yQtyToken == null ? null : ToDecimal(yQtyToken.Replace("Y", "").Replace(".00", ""));

            var dateToken = tokens.FirstOrDefault(x => DateTime.TryParse(x, out _));
            var dateRemoved = ToDate(dateToken);

            var document = tokens.FirstOrDefault(x =>
                x.StartsWith("YN", StringComparison.OrdinalIgnoreCase) ||
                x.StartsWith("YR", StringComparison.OrdinalIgnoreCase) ||
                x.StartsWith("YQ", StringComparison.OrdinalIgnoreCase));

            rows.Add(new ParsedRemovalRowDto
            {
                HullNo = hullNo,
                Erlid = ToDecimal(erlidText),
                Apl = tokens.ElementAtOrDefault(1),
                Nsn = tokens.ElementAtOrDefault(2),
                PartNumber = tokens.ElementAtOrDefault(3),
                DateRemoved = dateRemoved,
                DocumentNo = document,
                Qty = qty,
                Removed = dateRemoved != null ? "Y" : "N",
                RemovedBy = ExtractRemovedBy(tokens),
                Remarks = ExtractRemarks(tokens)
            });
        }

        return await SaveRowsAsync(file.FileName, "TXT", uploadedBy, rows);
    }

    private async Task<RemovalUploadResultDto> SaveRowsAsync(
        string fileName,
        string fileType,
        string uploadedBy,
        List<ParsedRemovalRowDto> rows)
    {
        await using var conn = GetConnection();
        await conn.OpenAsync();

        var batchId = await CreateBatchAsync(conn, fileName, fileType, uploadedBy);

        var matched = 0;
        var inserted = 0;
        var error = 0;

        foreach (var row in rows)
        {
            try
            {
                await using var cmd = conn.CreateCommand();
                cmd.CommandText = "PRTHRVST.PKG_PHT_REMOVAL_STATUS.UPSERT_REMOVAL_ROW";
                cmd.CommandType = CommandType.StoredProcedure;

                cmd.Parameters.Add("P_BATCH_ID", OracleDbType.Decimal).Value = batchId;
                cmd.Parameters.Add("P_HULL_NO", OracleDbType.Varchar2).Value = (object?)row.HullNo ?? DBNull.Value;
                cmd.Parameters.Add("P_ERLID", OracleDbType.Decimal).Value = (object?)row.Erlid ?? DBNull.Value;
                cmd.Parameters.Add("P_APL", OracleDbType.Varchar2).Value = (object?)row.Apl ?? DBNull.Value;
                cmd.Parameters.Add("P_APL_NOMEN", OracleDbType.Varchar2).Value = (object?)row.AplNomen ?? DBNull.Value;
                cmd.Parameters.Add("P_SAC_NOMEN", OracleDbType.Varchar2).Value = (object?)row.SacNomen ?? DBNull.Value;
                cmd.Parameters.Add("P_ONBOARD_LOCATION", OracleDbType.Varchar2).Value = (object?)row.OnBoardLocation ?? DBNull.Value;
                cmd.Parameters.Add("P_PART_NUMBER", OracleDbType.Varchar2).Value = (object?)row.PartNumber ?? DBNull.Value;
                cmd.Parameters.Add("P_NSN", OracleDbType.Varchar2).Value = (object?)row.Nsn ?? DBNull.Value;
                cmd.Parameters.Add("P_DWG_NO", OracleDbType.Varchar2).Value = (object?)row.DwgNo ?? DBNull.Value;
                cmd.Parameters.Add("P_QTY", OracleDbType.Decimal).Value = (object?)row.Qty ?? DBNull.Value;
                cmd.Parameters.Add("P_CAS_REP_IND", OracleDbType.Varchar2).Value = (object?)row.CasRepInd ?? DBNull.Value;
                cmd.Parameters.Add("P_IDENT", OracleDbType.Varchar2).Value = (object?)row.Ident ?? DBNull.Value;
                cmd.Parameters.Add("P_REMOVED", OracleDbType.Varchar2).Value = (object?)row.Removed ?? DBNull.Value;
                cmd.Parameters.Add("P_DATE_REMOVED", OracleDbType.Date).Value = (object?)row.DateRemoved ?? DBNull.Value;
                cmd.Parameters.Add("P_SYSTEM_ID", OracleDbType.Varchar2).Value = (object?)row.SystemId ?? DBNull.Value;
                cmd.Parameters.Add("P_FUND_CODE", OracleDbType.Varchar2).Value = (object?)row.FundCode ?? DBNull.Value;
                cmd.Parameters.Add("P_DOCUMENT_NO", OracleDbType.Varchar2).Value = (object?)row.DocumentNo ?? DBNull.Value;
                cmd.Parameters.Add("P_REMOVED_BY", OracleDbType.Varchar2).Value = (object?)row.RemovedBy ?? DBNull.Value;
                cmd.Parameters.Add("P_REMARKS", OracleDbType.Varchar2).Value = (object?)row.Remarks ?? DBNull.Value;
                cmd.Parameters.Add("P_UPDATED_BY", OracleDbType.Varchar2).Value = uploadedBy;

                var action = new OracleParameter("O_ACTION", OracleDbType.Varchar2, 50)
                {
                    Direction = ParameterDirection.Output
                };
                cmd.Parameters.Add(action);

                await cmd.ExecuteNonQueryAsync();

                var actionValue = action.Value?.ToString();

                if (actionValue == "MATCHED")
                    matched++;
                else if (actionValue == "INSERTED")
                    inserted++;
            }
            catch
            {
                error++;
            }
        }

        return new RemovalUploadResultDto
        {
            BatchId = batchId,
            TotalRows = rows.Count,
            MatchedRows = matched,
            InsertedRows = inserted,
            ErrorRows = error
        };
    }

    private async Task<decimal> CreateBatchAsync(OracleConnection conn, string fileName, string fileType, string uploadedBy)
    {
        await using var cmd = conn.CreateCommand();
        cmd.CommandText = "PRTHRVST.PKG_PHT_REMOVAL_STATUS.CREATE_UPLOAD_BATCH";
        cmd.CommandType = CommandType.StoredProcedure;

        cmd.Parameters.Add("P_FILE_NAME", OracleDbType.Varchar2).Value = fileName;
        cmd.Parameters.Add("P_FILE_TYPE", OracleDbType.Varchar2).Value = fileType;
        cmd.Parameters.Add("P_UPLOADED_BY", OracleDbType.Varchar2).Value = uploadedBy;

        var outParam = new OracleParameter("O_BATCH_ID", OracleDbType.Decimal)
        {
            Direction = ParameterDirection.Output
        };

        cmd.Parameters.Add(outParam);

        await cmd.ExecuteNonQueryAsync();

        return ((OracleDecimal)outParam.Value).Value;
    }

    private static string? GetCell(IXLRow row, Dictionary<string, int> headers, string header)
    {
        return headers.TryGetValue(header, out var col)
            ? row.Cell(col).GetString().Trim()
            : null;
    }

    private static decimal? ToDecimal(string? value)
    {
        return decimal.TryParse(value, out var result) ? result : null;
    }

    private static DateTime? ToDate(string? value)
    {
        return DateTime.TryParse(value, out var result) ? result : null;
    }

    private static string? ExtractRemovedBy(string[] tokens)
    {
        var erlIndex = Array.FindIndex(tokens, x => x.Equals("ERL", StringComparison.OrdinalIgnoreCase));

        if (erlIndex > 0)
            return tokens.ElementAtOrDefault(erlIndex - 1);

        return null;
    }

    private static string? ExtractRemarks(string[] tokens)
    {
        var index = Array.FindIndex(tokens, x => Regex.IsMatch(x, @"^\d{4}$"));

        if (index >= 0 && index + 1 < tokens.Length)
            return string.Join(" ", tokens.Skip(index + 1));

        return null;
    }

@Injectable({
  providedIn: 'root'

  private readonly http = inject(HttpClient);
  private readonly baseUrl = '/api/removal-status';

  getSummary(batchId?: number | null): Observable<RemovalSummary> {
    let params = new HttpParams();

    if (batchId) {
      params = params.set('batchId', batchId);
    }

    return this.http.get<RemovalSummary>(`${this.baseUrl}/summary`, { params });
  }

  getGrid(filters: {
    batchId?: number | null;
    systemId?: string | null;
    status?: string | null;

    let params = new HttpParams();

    if (filters.batchId) params = params.set('batchId', filters.batchId);
    if (filters.systemId) params = params.set('systemId', filters.systemId);
    if (filters.status) params = params.set('status', filters.status);
    if (filters.search) params = params.set('search', filters.search);

    return this.http.get<RemovalStatusRow[]>(`${this.baseUrl}/grid`, { params });
  }

  uploadExcel(file: File): Observable<RemovalUploadResult> {
    const formData = new FormData();
    formData.append('file', file);

    return this.http.post<RemovalUploadResult>(
      `${this.baseUrl}/upload-excel`,
      formData
    );
  }

    const formData = new FormData();
    formData.append('file', file);

      `${this.baseUrl}/upload-txt`,

    );
  }

import { CommonModule } from '@angular/common';
import { Component, OnInit, inject } from '@angular/core';
import { FormsModule } from '@angular/forms';
import {
  RemovalStatusRow,
  RemovalSummary,
  RemovalUploadResult
} from './removal-status.model';
import { RemovalStatusService } from './removal-status.service';

@Component({
  selector: 'app-removal-status',
  standalone: true,
  imports: [CommonModule, FormsModule],
  templateUrl: './removal-status.component.html',
  styleUrls: ['./removal-status.component.css']
})
export class RemovalStatusComponent implements OnInit {
  private readonly service = inject(RemovalStatusService);

  summary: RemovalSummary = {
    totalErlRequests: 0,
    inProcess: 0,
    removed: 0,
    discrepancies: 0
  };

  rows: RemovalStatusRow[] = [];

  selectedBatchId: number | null = null;
  systemId = '';
  status = '';
  search = '';

  selectedExcelFile: File | null = null;
  selectedTxtFile: File | null = null;

  loading = false;
  uploading = false;
  errorMessage = '';
  successMessage = '';

  lastUploadResult?: RemovalUploadResult;

  readonly statusOptions = [
    '',
    'In Process',
    'Removed',
    'Shipyard Only'
  ];

  ngOnInit(): void {
    this.loadAll();
  }

  loadAll(): void {
    this.loadSummary();
    this.loadGrid();
  }

  loadSummary(): void {
    this.service.getSummary(this.selectedBatchId).subscribe({
      next: result => {
        this.summary = result;
      },
      error: err => {
        console.error(err);
        this.errorMessage = 'Failed to load summary.';
      }
    });
  }

  loadGrid(): void {
    this.loading = true;
    this.errorMessage = '';

    this.service.getGrid({
      batchId: this.selectedBatchId,
      systemId: this.systemId || null,
      status: this.status || null,
      search: this.search || null
    }).subscribe({
      next: result => {
        this.rows = result;
        this.loading = false;
      },
      error: err => {
        console.error(err);
        this.errorMessage = 'Failed to load removal grid.';
        this.loading = false;
      }
    });
  }

  onExcelSelected(event: Event): void {
    const input = event.target as HTMLInputElement;
    this.selectedExcelFile = input.files?.[0] ?? null;
  }

  onTxtSelected(event: Event): void {
    const input = event.target as HTMLInputElement;
    this.selectedTxtFile = input.files?.[0] ?? null;
  }

  uploadExcel(): void {
    if (!this.selectedExcelFile) {
      this.errorMessage = 'Please select an Excel file first.';
      return;
    }

    this.uploading = true;
    this.errorMessage = '';
    this.successMessage = '';

    this.service.uploadExcel(this.selectedExcelFile).subscribe({
      next: result => {
        this.lastUploadResult = result;
        this.selectedBatchId = result.batchId;
        this.successMessage = `Excel uploaded. Matched: ${result.matchedRows}, Inserted: ${result.insertedRows}, Errors: ${result.errorRows}`;
        this.uploading = false;
        this.loadAll();
      },
      error: err => {
        console.error(err);
        this.errorMessage = 'Excel upload failed.';
        this.uploading = false;
      }
    });
  }

  uploadTxt(): void {
    if (!this.selectedTxtFile) {
      this.errorMessage = 'Please select a TXT file first.';
      return;
    }

    this.uploading = true;
    this.errorMessage = '';
    this.successMessage = '';

    this.service.uploadTxt(this.selectedTxtFile).subscribe({
      next: result => {
        this.lastUploadResult = result;
        this.selectedBatchId = result.batchId;
        this.successMessage = `TXT uploaded. Matched: ${result.matchedRows}, Inserted: ${result.insertedRows}, Errors: ${result.errorRows}`;
        this.uploading = false;
        this.loadAll();
      },
      error: err => {
        console.error(err);
        this.errorMessage = 'TXT upload failed.';
        this.uploading = false;
      }
    });
  }

  clearBatchFilter(): void {
    this.selectedBatchId = null;
    this.loadAll();
  }

  applyFilters(): void {
    this.loadAll();
  }

  getStatusClass(status?: string): string {
    if (!status) return 'status-default';

    if (status.toLowerCase().includes('removed')) return 'status-removed';
    if (status.toLowerCase().includes('process')) return 'status-process';
    if (status.toLowerCase().includes('shipyard')) return 'status-discrepancy';

    return 'status-default';
  }
}
