import { CommonModule } from '@angular/common';
import {
  Component,
  OnInit,
  computed,
  inject,
  signal
} from '@angular/core';
import { FormsModule } from '@angular/forms';

import {
  LocationBreakdown,
  RemovalDashboard,
  RemovalGridRow,
  RemovalUploadResult
} from './removal-status.models';

import { RemovalStatusService } from './removal-status.service';

@Component({
  selector: 'app-removal-status',
  standalone: true,
  imports: [
    CommonModule,
    FormsModule
  ],
  templateUrl: './removal-status.component.html',
  styleUrls: ['./removal-status.component.css']
})
export class RemovalStatusComponent implements OnInit {
  private readonly service = inject(RemovalStatusService);

  readonly locations = signal<string[]>([]);
  readonly selectedLocation = signal<string | null>(null);

  readonly dashboard = signal<RemovalDashboard>({
    summary: {
      totalStocked: 0,
      removedShipped: 0,
      completed: 0,
      inProcess: 0
    },
    locationBreakdown: []
  });

  readonly rows = signal<RemovalGridRow[]>([]);
  readonly totalCount = signal(0);
  readonly loading = signal(false);
  readonly uploading = signal(false);

  search = '';
  page = 1;
  pageSize = 50;

  showUploadPanel = false;
  uploadLocation = '';
  selectedFile: File | null = null;

  successMessage = '';
  errorMessage = '';
  uploadResult?: RemovalUploadResult;

  readonly donutStyle = computed(() => {
    const values = this.dashboard().locationBreakdown;

    const total = values.reduce(
      (sum, item) => sum + item.itemCount,
      0
    );

    if (total === 0) {
      return {
        background: '#e5e7eb'
      };
    }

    const colors = [
      '#1d4ed8',
      '#059669',
      '#f59e0b',
      '#7c3aed',
      '#dc2626',
      '#0891b2'
    ];

    let current = 0;

    const segments = values.map((item, index) => {
      const start = current;
      const end =
        current + (item.itemCount / total) * 100;

      current = end;

      return `${colors[index % colors.length]} ${start}% ${end}%`;
    });

    return {
      background: `conic-gradient(${segments.join(', ')})`
    };
  });

  readonly chartColors = [
    '#1d4ed8',
    '#059669',
    '#f59e0b',
    '#7c3aed',
    '#dc2626',
    '#0891b2'
  ];

  ngOnInit(): void {
    this.loadLocations();
    this.loadPage();
  }

  loadLocations(): void {
    this.service.getLocations().subscribe({
      next: locations => {
        this.locations.set(locations);
      },
      error: error => {
        console.error(error);
        this.errorMessage = 'Unable to load locations.';
      }
    });
  }

  loadPage(): void {
    this.loadDashboard();
    this.loadGrid();
  }

  loadDashboard(): void {
    this.service
      .getDashboard(this.selectedLocation())
      .subscribe({
        next: result => {
          this.dashboard.set(result);
        },
        error: error => {
          console.error(error);
          this.errorMessage =
            'Unable to load removal summary.';
        }
      });
  }

  loadGrid(): void {
    this.loading.set(true);

    this.service.getGrid({
      location: this.selectedLocation(),
      search: this.search,
      page: this.page,
      pageSize: this.pageSize
    }).subscribe({
      next: result => {
        this.rows.set(result.items);
        this.totalCount.set(result.totalCount);
        this.loading.set(false);
      },
      error: error => {
        console.error(error);
        this.errorMessage =
          'Unable to load removal-status records.';
        this.loading.set(false);
      }
    });
  }

  selectLocation(location: string | null): void {
    this.selectedLocation.set(location);
    this.page = 1;
    this.loadPage();
  }

  applySearch(): void {
    this.page = 1;
    this.loadGrid();
  }

  previousPage(): void {
    if (this.page <= 1) {
      return;
    }

    this.page--;
    this.loadGrid();
  }

  nextPage(): void {
    if (this.page * this.pageSize >= this.totalCount()) {
      return;
    }

    this.page++;
    this.loadGrid();
  }

  openUpload(): void {
    this.showUploadPanel = true;
    this.uploadLocation =
      this.selectedLocation() ?? '';
    this.selectedFile = null;
    this.successMessage = '';
    this.errorMessage = '';
    this.uploadResult = undefined;
  }

  closeUpload(): void {
    this.showUploadPanel = false;
  }

  onFileSelected(event: Event): void {
    const input = event.target as HTMLInputElement;

    this.selectedFile =
      input.files?.length
        ? input.files[0]
        : null;
  }

  upload(): void {
    if (!this.uploadLocation) {
      this.errorMessage =
        'Select a location before uploading.';
      return;
    }

    if (!this.selectedFile) {
      this.errorMessage =
        'Select an Excel file before uploading.';
      return;
    }

    this.uploading.set(true);
    this.successMessage = '';
    this.errorMessage = '';

    this.service
      .uploadExcel(
        this.selectedFile,
        this.uploadLocation
      )
      .subscribe({
        next: result => {
          this.uploadResult = result;

          this.successMessage =
            `${result.updatedRows} records updated. ` +
            `${result.noMatchRows} records did not match.`;

          this.uploading.set(false);

          this.selectedLocation.set(
            this.uploadLocation
          );

          this.loadLocations();
          this.loadPage();
        },
        error: error => {
          console.error(error);

          this.errorMessage =
            error?.error?.message
            ?? error?.error
            ?? 'Excel upload failed.';

          this.uploading.set(false);
        }
      });
  }

  percentage(item: LocationBreakdown): number {
    const total = this.dashboard()
      .locationBreakdown
      .reduce(
        (sum, row) => sum + row.itemCount,
        0
      );

    return total === 0
      ? 0
      : (item.itemCount / total) * 100;
  }
}

import { Injectable, inject } from '@angular/core';
import { HttpClient, HttpParams } from '@angular/common/http';
import { Observable } from 'rxjs';

import {
  PagedResult,
  RemovalDashboard,
  RemovalGridRow,
  RemovalUploadResult
} from './removal-status.models';

@Injectable({
  providedIn: 'root'
})
export class RemovalStatusService {
  private readonly http = inject(HttpClient);
  private readonly baseUrl = '/api/removal-status';

  getLocations(): Observable<string[]> {
    return this.http.get<string[]>(
      `${this.baseUrl}/locations`
    );
  }

  getDashboard(
    location?: string | null
  ): Observable<RemovalDashboard> {
    let params = new HttpParams();

    if (location) {
      params = params.set('location', location);
    }

    return this.http.get<RemovalDashboard>(
      `${this.baseUrl}/dashboard`,
      { params }
    );
  }

  getGrid(filters: {
    location?: string | null;
    search?: string | null;
    page: number;
    pageSize: number;
  }): Observable<PagedResult<RemovalGridRow>> {
    let params = new HttpParams()
      .set('page', filters.page)
      .set('pageSize', filters.pageSize);

    if (filters.location) {
      params = params.set('location', filters.location);
    }

    if (filters.search) {
      params = params.set('search', filters.search);
    }

    return this.http.get<PagedResult<RemovalGridRow>>(
      `${this.baseUrl}/grid`,
      { params }
    );
  }

  uploadExcel(
    file: File,
    location: string
  ): Observable<RemovalUploadResult> {
    const formData = new FormData();

    formData.append('file', file);
    formData.append('location', location);

    return this.http.post<RemovalUploadResult>(
      `${this.baseUrl}/upload`,
      formData
    );
  }
}

    public async Task<RemovalUploadResultDto> UploadAsync(
        IFormFile file,
        string location,
        string updatedBy,
        CancellationToken cancellationToken = default)
    {
        if (file.Length == 0)
            throw new InvalidOperationException("The Excel file is empty.");

        if (string.IsNullOrWhiteSpace(location))
            throw new InvalidOperationException(
                "A removal location must be selected.");

        var extension =
            Path.GetExtension(file.FileName).ToLowerInvariant();

        if (extension != ".xlsx")
        {
            throw new InvalidOperationException(
                "Only .xlsx Excel files are supported.");
        }

        var validRows = new List<RemovalExcelRowDto>();
        var errors = new List<RemovalUploadErrorDto>();

        using var stream = file.OpenReadStream();
        using var package = new ExcelPackage(stream);

        var worksheet =
            package.Workbook.Worksheets.FirstOrDefault();

        if (worksheet?.Dimension == null)
        {
            throw new InvalidOperationException(
                "The Excel workbook does not contain data.");
        }

        var headers = BuildHeaderMap(worksheet);

        ValidateHeaders(headers);

        for (
            var rowNumber = worksheet.Dimension.Start.Row + 1;
            rowNumber <= worksheet.Dimension.End.Row;
            rowNumber++)
        {
            if (IsEmptyRow(worksheet, rowNumber))
                continue;

            try
            {
                var row = ParseRow(
                    worksheet,
                    rowNumber,
                    headers);

                ValidateRow(row, rowNumber);

                validRows.Add(row);
            }
            catch (Exception ex)
            {
                errors.Add(new RemovalUploadErrorDto
                {
                    RowNumber = rowNumber,
                    Message = ex.Message
                });
            }
        }

        if (validRows.Count == 0)
        {
            return new RemovalUploadResultDto
            {
                TotalRows = errors.Count,
                InvalidRows = errors.Count,
                Errors = errors
            };
        }

        var databaseResult =
            await _repository.ProcessUploadAsync(
                location.Trim(),
                updatedBy,
                validRows,
                cancellationToken);

        databaseResult.TotalRows =
            validRows.Count + errors.Count;

        databaseResult.ValidRows = validRows.Count;
        databaseResult.InvalidRows = errors.Count;
        databaseResult.Errors = errors;

        return databaseResult;
    }

    private static RemovalExcelRowDto ParseRow(
        ExcelWorksheet worksheet,
        int rowNumber,
        IReadOnlyDictionary<string, int> headers)
    {
        return new RemovalExcelRowDto
        {
            Hull = GetCell(
                worksheet, rowNumber, headers, "HULL"),

            Niin = GetCell(
                worksheet, rowNumber, headers, "NIIN"),

            Apl = GetCell(
                worksheet, rowNumber, headers, "APL"),

            PartNumber = GetCell(
                worksheet, rowNumber, headers, "PARTNUMBER"),

            Erlid = ToDecimal(
                GetCell(
                    worksheet, rowNumber, headers, "ERLID")),

            Entered = GetCell(
                worksheet, rowNumber, headers, "ENTERED"),

            Assigned = GetCell(
                worksheet, rowNumber, headers, "ASSIGNED"),

            Removed = GetCell(
                worksheet, rowNumber, headers, "REMOVED"),

            OnBoardLocation = GetCell(
                worksheet,
                rowNumber,
                headers,
                "ONBOARDLOCATION"),

            FundCode = GetCell(
                worksheet, rowNumber, headers, "FUNDCODE"),

            Dd1149 = GetCell(
                worksheet, rowNumber, headers, "DD1149"),

            DateShip = ToIsoDate(
                worksheet,
                rowNumber,
                headers,
                "DATESHIP"),

            Complete = GetCell(
                worksheet, rowNumber, headers, "COMPLETE"),

            RemovedQty = ToDecimal(
                GetCell(
                    worksheet,
                    rowNumber,
                    headers,
                    "REMOVEDQTY")),

            InaType = GetCell(
                worksheet, rowNumber, headers, "INATYPE"),

            ShipToId = GetCell(
                worksheet, rowNumber, headers, "SHIPTOID"),

            ErlComment = GetCell(
                worksheet, rowNumber, headers, "ERLCOMMENT"),

            CasRepInd = GetCell(
                worksheet, rowNumber, headers, "CASREPIND"),

            ShipYard = GetCell(
                worksheet, rowNumber, headers, "SHIPYARD")
        };
    }

    private static Dictionary<string, int> BuildHeaderMap(
        ExcelWorksheet worksheet)
    {
        var headers =
            new Dictionary<string, int>(
                StringComparer.OrdinalIgnoreCase);

        var headerRow = worksheet.Dimension.Start.Row;

        for (
            var column = worksheet.Dimension.Start.Column;
            column <= worksheet.Dimension.End.Column;
            column++)
        {
            var rawHeader =
                worksheet.Cells[headerRow, column]
                    .Text
                    .Trim();

            if (string.IsNullOrWhiteSpace(rawHeader))
                continue;

            var normalized =
                NormalizeHeader(rawHeader);

            if (!headers.ContainsKey(normalized))
                headers.Add(normalized, column);
        }

        return headers;
    }

    private static void ValidateHeaders(
        IReadOnlyDictionary<string, int> headers)
    {
        var requiredHeaders = new[]
        {
            "HULL",
            "NIIN",
            "APL",
            "PARTNUMBER",
            "ERLID",
            "ENTERED",
            "ASSIGNED",
            "REMOVED",
            "ONBOARDLOCATION",
            "FUNDCODE",
            "DD1149",
            "DATESHIP",
            "COMPLETE",
            "REMOVEDQTY",
            "INATYPE",
            "SHIPTOID",
            "ERLCOMMENT",
            "CASREPIND",
            "SHIPYARD"
        };

        var missing = requiredHeaders
            .Where(header => !headers.ContainsKey(header))
            .ToArray();

        if (missing.Length > 0)
        {
            throw new InvalidOperationException(
                $"Missing Excel headers: {string.Join(", ", missing)}");
        }
    }

    private static void ValidateRow(
        RemovalExcelRowDto row,
        int rowNumber)
    {
        if (string.IsNullOrWhiteSpace(row.Hull))
            throw new InvalidOperationException(
                $"Row {rowNumber}: HULL is required.");

        if (string.IsNullOrWhiteSpace(row.Niin))
            throw new InvalidOperationException(
                $"Row {rowNumber}: NIIN is required.");

        if (string.IsNullOrWhiteSpace(row.Apl))
            throw new InvalidOperationException(
                $"Row {rowNumber}: APL is required.");

        if (string.IsNullOrWhiteSpace(row.PartNumber))
            throw new InvalidOperationException(
                $"Row {rowNumber}: PART_NUMBER is required.");
    }

    private static bool IsEmptyRow(
        ExcelWorksheet worksheet,
        int rowNumber)
    {
        for (
            var column = worksheet.Dimension.Start.Column;
            column <= worksheet.Dimension.End.Column;
            column++)
        {
            if (!string.IsNullOrWhiteSpace(
                    worksheet.Cells[rowNumber, column].Text))
            {
                return false;
            }
        }

        return true;
    }

    private static string? GetCell(
        ExcelWorksheet worksheet,
        int rowNumber,
        IReadOnlyDictionary<string, int> headers,
        string header)
    {
        if (!headers.TryGetValue(header, out var column))
            return null;

        var value =
            worksheet.Cells[rowNumber, column]
                .Text
                .Trim();

        return string.IsNullOrWhiteSpace(value)
            ? null
            : value;
    }

    private static string? ToIsoDate(
        ExcelWorksheet worksheet,
        int rowNumber,
        IReadOnlyDictionary<string, int> headers,
        string header)
    {
        if (!headers.TryGetValue(header, out var column))
            return null;

        var cell = worksheet.Cells[rowNumber, column];

        if (cell.Value is DateTime dateValue)
            return dateValue.ToString("yyyy-MM-dd");

        var text = cell.Text.Trim();

        if (string.IsNullOrWhiteSpace(text))
            return null;

        if (DateTime.TryParse(
                text,
                CultureInfo.InvariantCulture,
                DateTimeStyles.None,
                out var parsed))
        {
            return parsed.ToString("yyyy-MM-dd");
        }

        if (DateTime.TryParse(text, out parsed))
            return parsed.ToString("yyyy-MM-dd");

        throw new InvalidOperationException(
            $"Row {rowNumber}: DateShip '{text}' is not a valid date.");
    }

    private static decimal? ToDecimal(string? value)
    {
        if (string.IsNullOrWhiteSpace(value))
            return null;

        var normalized = value.Trim();

        if (normalized.EndsWith(".00"))
            normalized = normalized[..^3];

        return decimal.TryParse(
            normalized,
            NumberStyles.Any,
            CultureInfo.InvariantCulture,
            out var result)
            ? result
            : throw new InvalidOperationException(
                $"'{value}' is not a valid number.");
    }

    private static string NormalizeHeader(string value)
    {
        return new string(
            value
                .Where(char.IsLetterOrDigit)
                .Select(char.ToUpperInvariant)
                .ToArray());
    }
}

    private OracleConnection CreateConnection()
    {
        var connectionString =
            _configuration.GetConnectionString("OracleDb")
            ?? throw new InvalidOperationException(
                "OracleDb connection string is missing.");

        return new OracleConnection(connectionString);
    }

    public async Task<IReadOnlyList<string>> GetLocationsAsync(
        CancellationToken cancellationToken = default)
    {
        var results = new List<string>();

        await using var connection = CreateConnection();
        await connection.OpenAsync(cancellationToken);

        await using var command = connection.CreateCommand();
        command.BindByName = true;
        command.CommandType = CommandType.StoredProcedure;
        command.CommandText =
            "PRTHRVST.PKG_PHT_REMOVAL_STATUS.GET_LOCATIONS";

        command.Parameters.Add(
            "O_CURSOR",
            OracleDbType.RefCursor,
            ParameterDirection.Output);

        await using var reader =
            await command.ExecuteReaderAsync(cancellationToken);

        while (await reader.ReadAsync(cancellationToken))
        {
            results.Add(reader["LOCATION"]?.ToString() ?? string.Empty);
        }

        return results;
    }

    public async Task<RemovalSummaryDto> GetSummaryAsync(
        string? location,
        CancellationToken cancellationToken = default)
    {
        await using var connection = CreateConnection();
        await connection.OpenAsync(cancellationToken);

        await using var command = connection.CreateCommand();
        command.BindByName = true;
        command.CommandType = CommandType.StoredProcedure;
        command.CommandText =
            "PRTHRVST.PKG_PHT_REMOVAL_STATUS.GET_SUMMARY";

        command.Parameters.Add(
            "P_LOCATION",
            OracleDbType.Varchar2).Value =
            DbValue(location);

        command.Parameters.Add(
            "O_CURSOR",
            OracleDbType.RefCursor,
            ParameterDirection.Output);

        await using var reader =
            await command.ExecuteReaderAsync(cancellationToken);

        if (!await reader.ReadAsync(cancellationToken))
            return new RemovalSummaryDto();

        return new RemovalSummaryDto
        {
            TotalStocked = ToInt(reader["TOTAL_STOCKED"]),
            RemovedShipped = ToInt(reader["REMOVED_SHIPPED"]),
            Completed = ToInt(reader["COMPLETED"]),
            InProcess = ToInt(reader["IN_PROCESS"])
        };
    }

    public async Task<IReadOnlyList<LocationBreakdownDto>>
        GetLocationBreakdownAsync(
            CancellationToken cancellationToken = default)
    {

        await using var connection = CreateConnection();

    public async Task<PagedResultDto<RemovalGridRowDto>> GetGridAsync(
        string? location,
        string? search,
        int page,
        int pageSize,
        CancellationToken cancellationToken = default)
    {
        var results = new List<RemovalGridRowDto>();

        page = Math.Max(page, 1);
        pageSize = Math.Clamp(pageSize, 1, 500);

        var offset = (page - 1) * pageSize;

        await using var connection = CreateConnection();
        await connection.OpenAsync(cancellationToken);

        await using var command = connection.CreateCommand();
        command.BindByName = true;
        command.CommandType = CommandType.StoredProcedure;
        command.CommandText =
            "PRTHRVST.PKG_PHT_REMOVAL_STATUS.GET_GRID";

        command.Parameters.Add(
            "P_LOCATION",
            OracleDbType.Varchar2).Value =
            DbValue(location);

        command.Parameters.Add(
            "P_SEARCH",
            OracleDbType.Varchar2).Value =
            DbValue(search);

        command.Parameters.Add(
            "P_OFFSET",
            OracleDbType.Int32).Value = offset;

        command.Parameters.Add(
            "P_PAGE_SIZE",
            OracleDbType.Int32).Value = pageSize;

        var totalParameter = command.Parameters.Add(
            "O_TOTAL",
            OracleDbType.Int32);

        totalParameter.Direction = ParameterDirection.Output;

        command.Parameters.Add(
            "O_CURSOR",
            OracleDbType.RefCursor,
            ParameterDirection.Output);

        await using var reader =
            await command.ExecuteReaderAsync(cancellationToken);

        while (await reader.ReadAsync(cancellationToken))
        {
            results.Add(MapGridRow(reader));
        }

        return new PagedResultDto<RemovalGridRowDto>
        {
            TotalCount = ToInt(totalParameter.Value),
            Items = results
        };
    }

    public async Task<RemovalUploadResultDto> ProcessUploadAsync(
        string location,
        string updatedBy,
        IReadOnlyList<RemovalExcelRowDto> rows,
        CancellationToken cancellationToken = default)
    {
        var json = JsonSerializer.Serialize(
            rows,
            new JsonSerializerOptions
            {
                PropertyNamingPolicy = JsonNamingPolicy.CamelCase
            });

        await using var connection = CreateConnection();
        await connection.OpenAsync(cancellationToken);

        await using var command = connection.CreateCommand();
        command.BindByName = true;
        command.CommandType = CommandType.StoredProcedure;
        command.CommandText =
            "PRTHRVST.PKG_PHT_REMOVAL_STATUS.PROCESS_EXCEL_UPLOAD";

        command.Parameters.Add(
            "P_LOCATION",
            OracleDbType.Varchar2).Value = location;

        command.Parameters.Add(
            "P_UPDATED_BY",
            OracleDbType.Varchar2).Value = updatedBy;

        command.Parameters.Add(
            "P_ROWS_JSON",
            OracleDbType.Clob).Value = json;

        var totalParameter = command.Parameters.Add(
            "O_TOTAL_ROWS",
            OracleDbType.Int32);

        totalParameter.Direction = ParameterDirection.Output;

        var updatedParameter = command.Parameters.Add(
            "O_UPDATED_ROWS",
            OracleDbType.Int32);

        updatedParameter.Direction = ParameterDirection.Output;

        var noMatchParameter = command.Parameters.Add(
            "O_NO_MATCH_ROWS",
            OracleDbType.Int32);

        noMatchParameter.Direction = ParameterDirection.Output;

        await command.ExecuteNonQueryAsync(cancellationToken);

        return new RemovalUploadResultDto
        {
            TotalRows = ToInt(totalParameter.Value),
            ValidRows = rows.Count,
            UpdatedRows = ToInt(updatedParameter.Value),
            NoMatchRows = ToInt(noMatchParameter.Value)
        };
    }

    private static RemovalGridRowDto MapGridRow(
        OracleDataReader reader)
    {
        return new RemovalGridRowDto
        {
            Id = ToDecimal(reader["ID"]),
            Hull = ToText(reader["HULL_ID"]),
            Niin = ToText(reader["NIIN_ID"]),
            Apl = ToText(reader["APL"]),
            PartNumber = ToText(reader["PART_NUMBER"]),
            Erlid = ToNullableDecimal(reader["ERLID"]),
            Entered = ToText(reader["ENTERED"]),
            Assigned = ToText(reader["ASSIGNED"]),
            Removed = ToText(reader["REMOVED"]),
            OnBoardLocation = ToText(reader["ONBOARD_LOCATION"]),
            FundCode = ToText(reader["FUND_CODE"]),
            Dd1149 = ToText(reader["DD1149"]),
            DateShip = ToNullableDate(reader["DATE_SHIP"]),
            Complete = ToText(reader["COMPLETE"]),
            RemovedQty = ToNullableDecimal(reader["REMOVED_QTY"]),
            InaType = ToText(reader["INA_TYPE"]),
            ShipToId = ToText(reader["SHIP_TO_ID"]),
            ErlComment = ToText(reader["ERL_COMMENT"]),
            CasRepInd = ToText(reader["CAS_REP_IND"]),
            ShipYard = ToText(reader["SHIPYARD"]),
            RemovalLocation = ToText(reader["REMOVAL_LOCATION"])
        };
    }

    private static object DbValue(string? value)
    {
        return string.IsNullOrWhiteSpace(value)
            ? DBNull.Value
            : value.Trim();
    }

    private static string? ToText(object value)
    {
        return value == DBNull.Value ? null : value.ToString();
    }

    private static int ToInt(object value)
    {
        if (value is null || value == DBNull.Value)
            return 0;

        if (value is OracleDecimal oracleDecimal)
            return oracleDecimal.IsNull
                ? 0
                : oracleDecimal.ToInt32();

        return Convert.ToInt32(value);
    }

    private static decimal ToDecimal(object value)
    {
        return value == DBNull.Value
            ? 0
            : Convert.ToDecimal(value);
    }

    private static decimal? ToNullableDecimal(object value)
    {
        return value == DBNull.Value
            ? null
            : Convert.ToDecimal(value);
    }

    private static DateTime? ToNullableDate(object value)
    {
        return value == DBNull.Value
            ? null
            : Convert.ToDateTime(value);
    }

CREATE OR REPLACE PACKAGE BODY PRTHRVST.PKG_PHT_REMOVAL_STATUS AS

    PROCEDURE GET_LOCATIONS (
        O_CURSOR OUT SYS_REFCURSOR
    ) AS
    BEGIN
        OPEN O_CURSOR FOR
            SELECT DISTINCT REMOVAL_LOCATION AS LOCATION
            FROM PRTHRVST.PHT_MATERIAL_REQUEST_SUBMISSION
            WHERE IS_DELETED = 0
              AND REMOVAL_LOCATION IS NOT NULL
            ORDER BY REMOVAL_LOCATION;
    END GET_LOCATIONS;


    PROCEDURE GET_SUMMARY (
        P_LOCATION IN VARCHAR2,
        O_CURSOR   OUT SYS_REFCURSOR
    ) AS
    BEGIN
        OPEN O_CURSOR FOR
            SELECT
                COUNT(*) AS TOTAL_STOCKED,

                SUM(
                    CASE
                        WHEN UPPER(NVL(REMOVED, 'N'))
                             IN ('Y', 'YES', 'REMOVED', 'SHIPPED')
                          OR DATE_SHIP IS NOT NULL
                        THEN 1
                        ELSE 0
                    END
                ) AS REMOVED_SHIPPED,

                SUM(
                    CASE
                        WHEN UPPER(NVL(COMPLETE, 'N'))
                             IN ('Y', 'YES', 'TRUE', 'COMPLETE')
                        THEN 1
                        ELSE 0
                    END
                ) AS COMPLETED,

                SUM(
                    CASE
                        WHEN UPPER(NVL(REMOVED, 'N'))
                             NOT IN ('Y', 'YES', 'REMOVED', 'SHIPPED')
                         AND DATE_SHIP IS NULL
                         AND UPPER(NVL(COMPLETE, 'N'))
                             NOT IN ('Y', 'YES', 'TRUE', 'COMPLETE')
                        THEN 1
                        ELSE 0
                    END
                ) AS IN_PROCESS

            FROM PRTHRVST.PHT_MATERIAL_REQUEST_SUBMISSION
            WHERE IS_DELETED = 0
              AND (
                    P_LOCATION IS NULL
                    OR REMOVAL_LOCATION = P_LOCATION
                  );
    END GET_SUMMARY;


    PROCEDURE GET_LOCATION_BREAKDOWN (
        O_CURSOR OUT SYS_REFCURSOR
    ) AS
    BEGIN
        OPEN O_CURSOR FOR
            SELECT
                NVL(REMOVAL_LOCATION, 'Unassigned') AS LOCATION,
                COUNT(*) AS ITEM_COUNT
            FROM PRTHRVST.PHT_MATERIAL_REQUEST_SUBMISSION
            WHERE IS_DELETED = 0
            GROUP BY NVL(REMOVAL_LOCATION, 'Unassigned')
            ORDER BY ITEM_COUNT DESC;
    END GET_LOCATION_BREAKDOWN;


    PROCEDURE GET_GRID (
        P_LOCATION  IN VARCHAR2,
        P_SEARCH    IN VARCHAR2,
        P_OFFSET    IN NUMBER,
        P_PAGE_SIZE IN NUMBER,
        O_TOTAL     OUT NUMBER,
        O_CURSOR    OUT SYS_REFCURSOR
    ) AS
    BEGIN
        SELECT COUNT(*)
        INTO O_TOTAL
        FROM PRTHRVST.PHT_MATERIAL_REQUEST_SUBMISSION T
        WHERE T.IS_DELETED = 0
          AND (
                P_LOCATION IS NULL
                OR T.REMOVAL_LOCATION = P_LOCATION
              )
          AND (
                P_SEARCH IS NULL
                OR UPPER(TO_CHAR(T.ERLID)) LIKE '%' || UPPER(P_SEARCH) || '%'
                OR UPPER(TO_CHAR(T.HULL_ID)) LIKE '%' || UPPER(P_SEARCH) || '%'
                OR UPPER(TO_CHAR(T.NIIN_ID)) LIKE '%' || UPPER(P_SEARCH) || '%'
                OR UPPER(T.APL) LIKE '%' || UPPER(P_SEARCH) || '%'
                OR UPPER(T.PART_NUMBER) LIKE '%' || UPPER(P_SEARCH) || '%'
                OR UPPER(T.ONBOARD_LOCATION) LIKE '%' || UPPER(P_SEARCH) || '%'
                OR UPPER(T.ERL_COMMENT) LIKE '%' || UPPER(P_SEARCH) || '%'
              );

        OPEN O_CURSOR FOR
            SELECT
                T.ID,
                T.HULL_ID,
                T.NIIN_ID,
                T.APL,
                T.PART_NUMBER,
                T.ERLID,
                T.ENTERED,
                T.ASSIGNED,
                T.REMOVED,
                T.ONBOARD_LOCATION,
                T.FUND_CODE,
                T.DD1149,
                T.DATE_SHIP,
                T.COMPLETE,
                T.REMOVED_QTY,
                T.INA_TYPE,
                T.SHIP_TO_ID,
                T.ERL_COMMENT,
                T.CAS_REP_IND,
                T.SHIPYARD,
                T.REMOVAL_LOCATION
            FROM PRTHRVST.PHT_MATERIAL_REQUEST_SUBMISSION T
            WHERE T.IS_DELETED = 0
              AND (
                    P_LOCATION IS NULL
                    OR T.REMOVAL_LOCATION = P_LOCATION
                  )
              AND (
                    P_SEARCH IS NULL
                    OR UPPER(TO_CHAR(T.ERLID)) LIKE '%' || UPPER(P_SEARCH) || '%'
                    OR UPPER(TO_CHAR(T.HULL_ID)) LIKE '%' || UPPER(P_SEARCH) || '%'
                    OR UPPER(TO_CHAR(T.NIIN_ID)) LIKE '%' || UPPER(P_SEARCH) || '%'
                    OR UPPER(T.APL) LIKE '%' || UPPER(P_SEARCH) || '%'
                    OR UPPER(T.PART_NUMBER) LIKE '%' || UPPER(P_SEARCH) || '%'
                    OR UPPER(T.ONBOARD_LOCATION) LIKE '%' || UPPER(P_SEARCH) || '%'
                    OR UPPER(T.ERL_COMMENT) LIKE '%' || UPPER(P_SEARCH) || '%'
                  )
            ORDER BY T.ERLID
            OFFSET NVL(P_OFFSET, 0) ROWS
            FETCH NEXT NVL(P_PAGE_SIZE, 50) ROWS ONLY;
    END GET_GRID;


    PROCEDURE PROCESS_EXCEL_UPLOAD (
        P_LOCATION      IN VARCHAR2,
        P_UPDATED_BY    IN VARCHAR2,
        P_ROWS_JSON     IN CLOB,
        O_TOTAL_ROWS    OUT NUMBER,
        O_UPDATED_ROWS  OUT NUMBER,
        O_NO_MATCH_ROWS OUT NUMBER
    ) AS
    BEGIN
        SELECT COUNT(*)
        INTO O_TOTAL_ROWS
        FROM JSON_TABLE(
            P_ROWS_JSON,
            '$[*]'
            COLUMNS (
                HULL        VARCHAR2(100) PATH '$.hull',
                NIIN        VARCHAR2(100) PATH '$.niin',
                APL         VARCHAR2(100) PATH '$.apl',
                PART_NUMBER VARCHAR2(200) PATH '$.partNumber'
            )
        );

        SELECT COUNT(*)
        INTO O_NO_MATCH_ROWS
        FROM JSON_TABLE(
            P_ROWS_JSON,
            '$[*]'
            COLUMNS (
                HULL        VARCHAR2(100) PATH '$.hull',
                NIIN        VARCHAR2(100) PATH '$.niin',
                APL         VARCHAR2(100) PATH '$.apl',
                PART_NUMBER VARCHAR2(200) PATH '$.partNumber'
            )
        ) S
        WHERE NOT EXISTS (
            SELECT 1
            FROM PRTHRVST.PHT_MATERIAL_REQUEST_SUBMISSION T
            WHERE T.IS_DELETED = 0
              AND TRIM(TO_CHAR(T.HULL_ID)) = TRIM(S.HULL)
              AND TRIM(TO_CHAR(T.NIIN_ID)) = TRIM(S.NIIN)
              AND UPPER(TRIM(NVL(T.APL, 'X')))
                    = UPPER(TRIM(NVL(S.APL, 'X')))
              AND UPPER(TRIM(NVL(T.PART_NUMBER, 'X')))
                    = UPPER(TRIM(NVL(S.PART_NUMBER, 'X')))
        );

        MERGE INTO PRTHRVST.PHT_MATERIAL_REQUEST_SUBMISSION T
        USING (
            SELECT *
            FROM (
                SELECT
                    J.*,
                    ROW_NUMBER() OVER (
                        PARTITION BY
                            TRIM(HULL),
                            TRIM(NIIN),
                            UPPER(TRIM(APL)),
                            UPPER(TRIM(PART_NUMBER))
                        ORDER BY ERLID
                    ) AS RN
                FROM JSON_TABLE(
                    P_ROWS_JSON,
                    '$[*]'
                    COLUMNS (
                        HULL               VARCHAR2(100)  PATH '$.hull',
                        NIIN               VARCHAR2(100)  PATH '$.niin',
                        APL                VARCHAR2(100)  PATH '$.apl',
                        PART_NUMBER        VARCHAR2(200)  PATH '$.partNumber',
                        ERLID              NUMBER         PATH '$.erlid',
                        ENTERED            VARCHAR2(100)  PATH '$.entered',
                        ASSIGNED           VARCHAR2(100)  PATH '$.assigned',
                        REMOVED            VARCHAR2(30)   PATH '$.removed',
                        ONBOARD_LOCATION   VARCHAR2(200)  PATH '$.onBoardLocation',
                        FUND_CODE          VARCHAR2(100)  PATH '$.fundCode',
                        DD1149             VARCHAR2(100)  PATH '$.dd1149',
                        DATE_SHIP_TEXT     VARCHAR2(50)   PATH '$.dateShip',
                        COMPLETE           VARCHAR2(30)   PATH '$.complete',
                        REMOVED_QTY        NUMBER         PATH '$.removedQty',
                        INA_TYPE           VARCHAR2(100)  PATH '$.inaType',
                        SHIP_TO_ID         VARCHAR2(100)  PATH '$.shipToId',
                        ERL_COMMENT        VARCHAR2(2000) PATH '$.erlComment',
                        CAS_REP_IND        VARCHAR2(50)   PATH '$.casRepInd',
                        SHIPYARD           VARCHAR2(200)  PATH '$.shipYard'
                    )
                ) J
            )
            WHERE RN = 1
        ) S
        ON (
            TRIM(TO_CHAR(T.HULL_ID)) = TRIM(S.HULL)
            AND TRIM(TO_CHAR(T.NIIN_ID)) = TRIM(S.NIIN)
            AND UPPER(TRIM(NVL(T.APL, 'X')))
                = UPPER(TRIM(NVL(S.APL, 'X')))
            AND UPPER(TRIM(NVL(T.PART_NUMBER, 'X')))
                = UPPER(TRIM(NVL(S.PART_NUMBER, 'X')))
            AND T.IS_DELETED = 0
        )
        WHEN MATCHED THEN
            UPDATE SET
                T.ERLID = S.ERLID,
                T.ENTERED = S.ENTERED,
                T.ASSIGNED = S.ASSIGNED,
                T.REMOVED = S.REMOVED,
                T.ONBOARD_LOCATION = S.ONBOARD_LOCATION,
                T.FUND_CODE = S.FUND_CODE,
                T.DD1149 = S.DD1149,
                T.DATE_SHIP =
                    CASE
                        WHEN S.DATE_SHIP_TEXT IS NULL THEN NULL
                        ELSE TO_DATE(
                            SUBSTR(S.DATE_SHIP_TEXT, 1, 10),
                            'YYYY-MM-DD'
                        )
                    END,
                T.COMPLETE = S.COMPLETE,
                T.REMOVED_QTY = S.REMOVED_QTY,
                T.INA_TYPE = S.INA_TYPE,
                T.SHIP_TO_ID = S.SHIP_TO_ID,
                T.ERL_COMMENT = S.ERL_COMMENT,
                T.CAS_REP_IND = S.CAS_REP_IND,
                T.SHIPYARD = S.SHIPYARD,
                T.REMOVAL_LOCATION = P_LOCATION,
                T.REMOVAL_UPLOAD_DT = SYSTIMESTAMP,
                T.REMOVAL_UPDATED_BY = P_UPDATED_BY,
                T.UPDATED_DT = SYSTIMESTAMP,
                T.UPDATED_BY = P_UPDATED_BY,
                T.STATUS =
                    CASE
                        WHEN UPPER(NVL(S.COMPLETE, 'N'))
                             IN ('Y', 'YES', 'TRUE', 'COMPLETE')
                          OR UPPER(NVL(S.REMOVED, 'N'))
                             IN ('Y', 'YES', 'REMOVED', 'SHIPPED')
                        THEN 'Removed'
                        ELSE T.STATUS
                    END;

        O_UPDATED_ROWS := SQL%ROWCOUNT;

        COMMIT;

    EXCEPTION
        WHEN OTHERS THEN
            ROLLBACK;
            RAISE;
    END PROCESS_EXCEL_UPLOAD;

END PKG_PHT_REMOVAL_STATUS;
