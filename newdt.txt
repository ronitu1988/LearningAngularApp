using Microsoft.AspNetCore.Mvc;
using PHTool.Api.Dtos;
using PHTool.Api.Services;

namespace PHTool.Api.Controllers;

[ApiController]
[Route("api/removal-upload")]
public class RemovalUploadController : ControllerBase
{
    private readonly IRemovalUploadService _service;

    public RemovalUploadController(IRemovalUploadService service)
    {
        _service = service;
    }

    [HttpPost("excel")]
    public async Task<ActionResult<RemovalUploadResultDto>> UploadExcel(IFormFile file)
    {
        if (file == null || file.Length == 0)
            return BadRequest("Excel file is required.");

        var extension = Path.GetExtension(file.FileName).ToLowerInvariant();

        if (extension != ".xlsx" && extension != ".xls")
            return BadRequest("Only .xlsx or .xls files are supported.");

        var updatedBy = User?.Identity?.Name ?? "SYSTEM";

        var result = await _service.UploadRemovalExcelAsync(file, updatedBy);

        return Ok(result);
    }
}



using ClosedXML.Excel;
using PHTool.Api.Data;
using PHTool.Api.Dtos;

namespace PHTool.Api.Services;

public class RemovalUploadService : IRemovalUploadService
{
    private readonly IRemovalUploadRepository _repository;

    public RemovalUploadService(IRemovalUploadRepository repository)
    {
        _repository = repository;
    }

    public async Task<RemovalUploadResultDto> UploadRemovalExcelAsync(IFormFile file, string updatedBy)
    {
        var result = new RemovalUploadResultDto();

        using var stream = file.OpenReadStream();
        using var workbook = new XLWorkbook(stream);

        var worksheet = workbook.Worksheets.First();
        var headerRow = worksheet.FirstRowUsed();

        if (headerRow == null)
            throw new InvalidOperationException("Excel file is empty.");

        var headers = headerRow.CellsUsed()
            .ToDictionary(
                c => NormalizeHeader(c.GetString()),
                c => c.Address.ColumnNumber
            );

        ValidateHeaders(headers);

        foreach (var row in worksheet.RowsUsed().Skip(1))
        {
            var rowNumber = row.RowNumber();
            result.TotalRows++;

            try
            {
                var dto = new RemovalUploadRowDto
                {
                    Hull = GetCell(row, headers, "HULL"),
                    Niin = GetCell(row, headers, "NIIN"),
                    Apl = GetCell(row, headers, "APL"),
                    PartNumber = GetCell(row, headers, "PARTNUMBER"),
                    Erlid = ToDecimal(GetCell(row, headers, "ERLID")),
                    Entered = GetCell(row, headers, "ENTERED"),
                    Assigned = GetCell(row, headers, "ASSIGNED"),
                    Removed = GetCell(row, headers, "REMOVED"),
                    OnBoardLocation = GetCell(row, headers, "ONBOARDLOCATION"),
                    FundCode = GetCell(row, headers, "FUNDCODE"),
                    Dd1149 = GetCell(row, headers, "DD1149"),
                    DateShip = ToDate(GetCell(row, headers, "DATESHIP")),
                    Complete = GetCell(row, headers, "COMPLETE"),
                    RemovedQty = ToDecimal(GetCell(row, headers, "REMOVEDQTY")),
                    InaType = GetCell(row, headers, "INATYPE"),
                    ShipToId = GetCell(row, headers, "SHIPTOID"),
                    ErlComment = GetCell(row, headers, "ERLCOMMENT"),
                    CasRepInd = GetCell(row, headers, "CASREPIND"),
                    ShipYard = GetCell(row, headers, "SHIPYARD")
                };

                ValidateRow(dto, rowNumber);

                var action = await _repository.UpsertRemovalUploadRowAsync(dto, updatedBy);

                if (action == "UPDATED")
                    result.UpdatedRows++;
                else if (action == "NO_MATCH")
                    result.NoMatchRows++;
                else
                    result.ErrorRows++;

                result.RowResults.Add(new RemovalUploadRowResultDto
                {
                    RowNumber = rowNumber,
                    Hull = dto.Hull,
                    Niin = dto.Niin,
                    Apl = dto.Apl,
                    PartNumber = dto.PartNumber,
                    Erlid = dto.Erlid,
                    Action = action
                });
            }
            catch (Exception ex)
            {
                result.ErrorRows++;

                result.RowResults.Add(new RemovalUploadRowResultDto
                {
                    RowNumber = rowNumber,
                    Action = "ERROR",
                    ErrorMessage = ex.Message
                });
            }
        }

        return result;
    }

    private static void ValidateHeaders(Dictionary<string, int> headers)
    {
        var requiredHeaders = new[]
        {
            "HULL",
            "NIIN",
            "APL",
            "PARTNUMBER",
            "ERLID",
            "ENTERED",
            "ASSIGNED",
            "REMOVED",
            "ONBOARDLOCATION",
            "FUNDCODE",
            "DD1149",
            "DATESHIP",
            "COMPLETE",
            "REMOVEDQTY",
            "INATYPE",
            "SHIPTOID",
            "ERLCOMMENT",
            "CASREPIND",
            "SHIPYARD"
        };

        var missing = requiredHeaders
            .Where(h => !headers.ContainsKey(h))
            .ToList();

        if (missing.Any())
            throw new InvalidOperationException($"Missing required Excel headers: {string.Join(", ", missing)}");
    }

    private static void ValidateRow(RemovalUploadRowDto row, int rowNumber)
    {
        if (string.IsNullOrWhiteSpace(row.Hull))
            throw new InvalidOperationException($"Row {rowNumber}: HULL is required.");

        if (string.IsNullOrWhiteSpace(row.Niin))
            throw new InvalidOperationException($"Row {rowNumber}: NIIN is required.");

        if (string.IsNullOrWhiteSpace(row.Apl))
            throw new InvalidOperationException($"Row {rowNumber}: APL is required.");

        if (string.IsNullOrWhiteSpace(row.PartNumber))
            throw new InvalidOperationException($"Row {rowNumber}: PART_NUMBER is required.");
    }

    private static string NormalizeHeader(string value)
    {
        return value
            .Replace(" ", "")
            .Replace("_", "")
            .Replace("-", "")
            .Trim()
            .ToUpperInvariant();
    }

    private static string? GetCell(IXLRow row, Dictionary<string, int> headers, string headerKey)
    {
        if (!headers.TryGetValue(headerKey, out var col))
            return null;

        var value = row.Cell(col).GetFormattedString()?.Trim();
        return string.IsNullOrWhiteSpace(value) ? null : value;
    }

    private static decimal? ToDecimal(string? value)
    {
        if (string.IsNullOrWhiteSpace(value))
            return null;

        value = value.Trim();

        if (value.EndsWith(".00"))
            value = value[..^3];

        return decimal.TryParse(value, out var result)
            ? result
            : null;
    }

    private static DateTime? ToDate(string? value)
    {
        if (string.IsNullOrWhiteSpace(value))
            return null;

        return DateTime.TryParse(value, out var result)
            ? result
            : null;
    }
}

using System.Data;
using Oracle.ManagedDataAccess.Client;
using PHTool.Api.Dtos;

namespace PHTool.Api.Data;

public class RemovalUploadRepository : IRemovalUploadRepository
{
    private readonly IConfiguration _configuration;

    public RemovalUploadRepository(IConfiguration configuration)
    {
        _configuration = configuration;
    }

    private OracleConnection GetConnection()
    {
        return new OracleConnection(_configuration.GetConnectionString("OracleDb"));
    }

    public async Task<string> UpsertRemovalUploadRowAsync(RemovalUploadRowDto row, string updatedBy)
    {
        await using var conn = GetConnection();
        await conn.OpenAsync();

        await using var cmd = conn.CreateCommand();
        cmd.CommandText = "PRTHRVST.PKG_PHT_REMOVAL_UPLOAD.UPSERT_REMOVAL_UPLOAD_ROW";
        cmd.CommandType = CommandType.StoredProcedure;
        cmd.BindByName = true;

        cmd.Parameters.Add("P_HULL", OracleDbType.Varchar2).Value = (object?)row.Hull ?? DBNull.Value;
        cmd.Parameters.Add("P_NIIN", OracleDbType.Varchar2).Value = (object?)row.Niin ?? DBNull.Value;
        cmd.Parameters.Add("P_APL", OracleDbType.Varchar2).Value = (object?)row.Apl ?? DBNull.Value;
        cmd.Parameters.Add("P_PART_NUMBER", OracleDbType.Varchar2).Value = (object?)row.PartNumber ?? DBNull.Value;
        cmd.Parameters.Add("P_ERLID", OracleDbType.Decimal).Value = (object?)row.Erlid ?? DBNull.Value;
        cmd.Parameters.Add("P_ENTERED", OracleDbType.Varchar2).Value = (object?)row.Entered ?? DBNull.Value;
        cmd.Parameters.Add("P_ASSIGNED", OracleDbType.Varchar2).Value = (object?)row.Assigned ?? DBNull.Value;
        cmd.Parameters.Add("P_REMOVED", OracleDbType.Varchar2).Value = (object?)row.Removed ?? DBNull.Value;
        cmd.Parameters.Add("P_ONBOARD_LOCATION", OracleDbType.Varchar2).Value = (object?)row.OnBoardLocation ?? DBNull.Value;
        cmd.Parameters.Add("P_FUND_CODE", OracleDbType.Varchar2).Value = (object?)row.FundCode ?? DBNull.Value;
        cmd.Parameters.Add("P_DD1149", OracleDbType.Varchar2).Value = (object?)row.Dd1149 ?? DBNull.Value;
        cmd.Parameters.Add("P_DATE_SHIP", OracleDbType.Date).Value = (object?)row.DateShip ?? DBNull.Value;
        cmd.Parameters.Add("P_COMPLETE", OracleDbType.Varchar2).Value = (object?)row.Complete ?? DBNull.Value;
        cmd.Parameters.Add("P_REMOVED_QTY", OracleDbType.Decimal).Value = (object?)row.RemovedQty ?? DBNull.Value;
        cmd.Parameters.Add("P_INA_TYPE", OracleDbType.Varchar2).Value = (object?)row.InaType ?? DBNull.Value;
        cmd.Parameters.Add("P_SHIP_TO_ID", OracleDbType.Varchar2).Value = (object?)row.ShipToId ?? DBNull.Value;
        cmd.Parameters.Add("P_ERL_COMMENT", OracleDbType.Varchar2).Value = (object?)row.ErlComment ?? DBNull.Value;
        cmd.Parameters.Add("P_CAS_REP_IND", OracleDbType.Varchar2).Value = (object?)row.CasRepInd ?? DBNull.Value;
        cmd.Parameters.Add("P_SHIPYARD", OracleDbType.Varchar2).Value = (object?)row.ShipYard ?? DBNull.Value;
        cmd.Parameters.Add("P_UPDATED_BY", OracleDbType.Varchar2).Value = updatedBy;

        var actionParam = new OracleParameter("O_ACTION", OracleDbType.Varchar2, 50)
        {
            Direction = ParameterDirection.Output
        };

        cmd.Parameters.Add(actionParam);

        await cmd.ExecuteNonQueryAsync();

        return actionParam.Value?.ToString() ?? "UNKNOWN";
    }
}

CREATE OR REPLACE PACKAGE BODY PRTHRVST.PKG_PHT_REMOVAL_UPLOAD AS

    PROCEDURE UPSERT_REMOVAL_UPLOAD_ROW (
        P_HULL              IN VARCHAR2,
        P_NIIN              IN VARCHAR2,
        P_APL               IN VARCHAR2,
        P_PART_NUMBER       IN VARCHAR2,
        P_ERLID             IN NUMBER,
        P_ENTERED           IN VARCHAR2,
        P_ASSIGNED          IN VARCHAR2,
        P_REMOVED           IN VARCHAR2,
        P_ONBOARD_LOCATION  IN VARCHAR2,
        P_FUND_CODE         IN VARCHAR2,
        P_DD1149            IN VARCHAR2,
        P_DATE_SHIP         IN DATE,
        P_COMPLETE          IN VARCHAR2,
        P_REMOVED_QTY       IN NUMBER,
        P_INA_TYPE          IN VARCHAR2,
        P_SHIP_TO_ID        IN VARCHAR2,
        P_ERL_COMMENT       IN VARCHAR2,
        P_CAS_REP_IND       IN VARCHAR2,
        P_SHIPYARD          IN VARCHAR2,
        P_UPDATED_BY        IN VARCHAR2,
        O_ACTION            OUT VARCHAR2
    ) AS
        V_ID NUMBER;
    BEGIN
        BEGIN
            SELECT ID
            INTO V_ID
            FROM PRTHRVST.PHT_MATERIAL_REQUEST_SUBMISSION
            WHERE TO_CHAR(HULL_ID) = TRIM(P_HULL)
              AND TO_CHAR(NIIN_ID) = TRIM(P_NIIN)
              AND NVL(TRIM(APL), 'X') = NVL(TRIM(P_APL), 'X')
              AND NVL(TRIM(PART_NUMBER), 'X') = NVL(TRIM(P_PART_NUMBER), 'X')
              AND IS_DELETED = 0
            FETCH FIRST 1 ROW ONLY;

            UPDATE PRTHRVST.PHT_MATERIAL_REQUEST_SUBMISSION
            SET
                ERLID = P_ERLID,
                ENTERED = P_ENTERED,
                ASSIGNED = P_ASSIGNED,
                REMOVED = P_REMOVED,
                ONBOARD_LOCATION = P_ONBOARD_LOCATION,
                FUND_CODE = P_FUND_CODE,
                DD1149 = P_DD1149,
                DATE_SHIP = P_DATE_SHIP,
                COMPLETE = P_COMPLETE,
                REMOVED_QTY = P_REMOVED_QTY,
                INA_TYPE = P_INA_TYPE,
                SHIP_TO_ID = P_SHIP_TO_ID,
                ERL_COMMENT = P_ERL_COMMENT,
                CAS_REP_IND = P_CAS_REP_IND,
                SHIPYARD = P_SHIPYARD,
                STATUS =
                    CASE
                        WHEN UPPER(NVL(P_COMPLETE, 'N')) IN ('Y', 'YES', 'TRUE', 'COMPLETE')
                          OR UPPER(NVL(P_REMOVED, 'N')) IN ('Y', 'YES', 'TRUE', 'REMOVED')
                        THEN 'Removed'
                        ELSE STATUS
                    END,
                REMOVAL_UPLOAD_DT = SYSTIMESTAMP,
                REMOVAL_UPDATED_BY = P_UPDATED_BY,
                UPDATED_DT = SYSTIMESTAMP,
                UPDATED_BY = P_UPDATED_BY
            WHERE ID = V_ID;

            O_ACTION := 'UPDATED';

        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                O_ACTION := 'NO_MATCH';
        END;
    END UPSERT_REMOVAL_UPLOAD_ROW;

END PKG_PHT_REMOVAL_UPLOAD;
